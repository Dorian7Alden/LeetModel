Python-PuLP 实现：import pulp

model = pulp.LpProblem(&#34;Loan&#34;, pulp.LpMaximize)

x1 = pulp.LpVariable(&#34;x1&#34;, lowBound=0)
x2 = pulp.LpVariable(&#34;x2&#34;, lowBound=0)
x3 = pulp.LpVariable(&#34;x3&#34;, lowBound=0)

# 目标函数
model += 0.12*x1 + 0.10*x2 + 0.15*x3

# 约束条件
model += x1 + x2 + x3 &lt;= 1000
model += 0.02*x1 + 0.03*x2 + 0.05*x3 &lt;= 30

model.solve()
print(&#34;最优解：&#34;, x1.value(), x2.value(), x3.value())

print(df.groupby(&#34;label&#34;).mean())
