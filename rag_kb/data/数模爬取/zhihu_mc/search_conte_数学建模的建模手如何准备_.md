from scipy.stats import norm

# ===================== 1. 论文公式参数配置（严格复刻）=====================
alpha = 0.02    # 中期临床风险系数
beta  = 0.05    # 晚期临床风险系数
kappa = 1.00    # 检测误差缩放因子
q     = 0.95    # 最低保障率约束

# Y染色体浓度拟合系数（临床常规拟合值，可后续用真实数据回归替换）
a0, a1, a2 = 2.0, 0.25, 0.08

# ===================== 2. 论文原版公式函数定义 =====================
# 1. 临床时间代价 phi(t)
def phi(t):
if t &lt;= 12:
return 0.0
elif 12 &lt; t &lt; 28:
return alpha * (t - 12)
else:
return 16 * alpha + beta * (t - 28)

# 2. 胎儿Y染色体理论浓度 mu(t,b)
def mu_conc(t, bmi):
return a0 + a1 * t - a2 * bmi

# 3. 带噪声的达标概率 p_kappa
def p_kappa(t, bmi):
mu = mu_conc(t, bmi)
z = (mu - 4.0) / kappa
return norm.cdf(z)

# 4. 检测失败风险
def r_fail(t, bmi):
return 1.0 - p_kappa(t, bmi)

# 5. 个体总风险（论文核心总风险公式）
def total_risk(t, bmi):
return r_fail(t, bmi) + phi(t)

# ===================== 3. 离散化（严格按论文做法）=====================
# BMI离散：20 ~ 47，步长0.1
bmi_values = np.arange(20.0, 47.0, 0.1)
# 孕周离散：10 ~ 28，步长0.1
week_values = np.arange(10.0, 28.1, 0.1)

n = len(bmi_values)
K = 5    # 固定分5组

# ===================== 4. 预计算任意BMI区间[i,j]最优孕周和最小风险 =====================
INF = 1e9
cost  = np.full((n, n), INF)   # 区间[i,j]最小总风险
best_t = np.zeros((n, n))      # 区间[i,j]对应最优检测孕周

for i in range(n):
for j in range(i, n):
min_avg_risk = INF
opt_w = 12.0
# 遍历所有候选孕周
for w in week_values:
# 约束：组内BMI最大者达标率 &gt;= q 0.95
bmi_max_in_group = bmi_values[j]
if p_kappa(w, bmi_max_in_group) &lt; q:
continue

# 计算区间平均总风险
sum_risk = 0.0
for k in range(i, j+1):
sum_risk += total_risk(w, bmi_values[k])
avg_risk = sum_risk / (j - i + 1)

if avg_risk &lt; min_avg_risk:
min_avg_risk = avg_risk
opt_w = w

cost[i, j] = min_avg_risk
best_t[i, j] = opt_w

# ===================== 5. 动态规划最优分组求解 =====================
# DP[j][s]：前j个BMI离散点分成s组的最小总风险
DP = np.full((n, K+1), INF)
prev = np.zeros((n, K+1), dtype=int)   # 回溯标记

# 初始化：分成1组
for j in range(n):
DP[j, 1] = cost[0, j]
prev[j, 1] = 0

# 状态递推：分2~5组
for s in range(2, K+1):
for j in range(s-1, n):
for i in range(s-2, j):
if DP[i, s-1] + cost[i+1, j] &lt; DP[j, s]:
DP[j, s] = DP[i, s-1] + cost[i+1, j]
prev[j, s] = i

# ===================== 6. 回溯得到最优BMI分组 &amp; 最佳检测孕周 =====================
groups = []
cur_idx = n - 1

for s in range(K, 0, -1):
split_idx = prev[cur_idx, s]
bmi_left  = bmi_values[split_idx]
bmi_right = bmi_values[cur_idx]
best_week = best_t[split_idx, cur_idx]
groups.append([bmi_left, bmi_right, best_week])
cur_idx = split_idx

# 正序排列
groups.reverse()

# ===================== 7. 打印论文格式结果 =====================
print(&#34;=&#34;*60)
print(&#34;        BMI最优分组及NIPT最佳检测时点（复刻论文公式）&#34;)
print(&#34;=&#34;*60)
print(f&#34;{&#39;组别&#39;:&lt;5}{&#39;BMI区间&#39;:&lt;15}{&#39;推荐检测孕周(周)&#39;:&lt;15}&#34;)
print(&#34;-&#34;*60)
for idx, (b1, b2, w) in enumerate(groups, 1):
print(f&#34;{idx:&lt;5}[{b1:.1f}, {b2:.1f})     {w:.1f}&#34;)
