## 赛事管理




```sql

CREATE TABLE `competition` (
  `competition_id` INT AUTO_INCREMENT PRIMARY KEY COMMENT '赛事ID',
  `competition_name` VARCHAR(100) NOT NULL COMMENT '赛事名称',
  `competition_description` TEXT COMMENT '赛事描述',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='赛事信息表';


```
